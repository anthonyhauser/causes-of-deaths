\headerbox{Background}{name=background,span=3,column=0,row=0}{

\begin{itemize}[leftmargin=1em]
\compresslist
    \item The COVID-19 pandemic reshaped mortality through:
\begin{itemize}
\compresslist
    \item Direct deaths from SARS-CoV-2,
    \item Associated deaths from other causes (e.g., cardiovascular diseases) triggered by infection,
    \item Mortality displacement across time and causes,
    \item Indirect effects of control measures and behavioural changes (e.g., fewer respiratory infections, healthcare disruptions).
\end{itemize}
\item These disruptions occur simultaneously across multiple causes, limiting the interpretability of all-cause and single-cause excess mortality.
\item A multivariate approach is necessary to capture cross-cause dependencies and facilitate interpretation of these disruptions.
\end{itemize}
}

\headerbox{Objectives}
{name=objectives,column=0,span=3,below=background}{
\begin{itemize}[leftmargin=1em]
\compresslist
\item Implement a Bayesian multivariate model to estimate cause-specific excess mortality over time in Switzerland.
\item Quantify dependencies between mortality causes before and during COVID-19.
\item Identify the mechanisms driving mortality disruptions during the pandemic.
\end{itemize}
}

\headerbox{Data}{name=data,column=0,span=3,below=objectives}{
\begin{itemize}[leftmargin=1em]
\compresslist
\item \textbf{Cause-specific deaths}: $y_{irs}(t)$
\begin{itemize}
\compresslist
    \item aggregated by cause $i$, sex $s$, NUTS-2 region $r$, year and week $t$; by age group (0–17, 18–39, 40–64, 65–79, 80+); in 2011-2021,
    \item based on the underlying cause of death (ICD-10),
    \item grouped into 9 categories: cardiovascular (CVD), mental/neurological, respiratory, infectious diseases, cancer, external causes, suicide, COVID-19 and others.
\end{itemize}
\item \textbf{Population sizes}: $N_{rst}$ by age, sex, region; weekly estimates via linear interpolation.
\end{itemize}
}

\headerbox{Model}{name=model,span=3,column=0,below=data}{
\begin{itemize}[leftmargin=1em]
\compresslist
    \item\textbf{Expected mortality}: 
For each age group, we fit the number of deaths $y_{irs}(t)$ due to underlying cause $i$ at year and week $t$, in region $r$ and in sex $s$ between 2011 and 2019: %\\[-0.2cm]
$$y_{irs}(t) \sim \text{Poisson}(\mu_{irs}(t))$$
with $\log \mu_{irs}(t) = \alpha_i + \beta_{ir} + \gamma_{is} + f_{i}(t) + g_{i}(t) + \varepsilon_i(t) + \log N_{rst}$ and where 
\begin{itemize}\compresslist
\item $\alpha_i$, $\beta_{ir}$, $\gamma_{is}$: baseline mortality, effects of region and sex,
\item $f_{i}(t)$, $g_{i}(t)$: long-term and seasonal trends (Gaussian processes),
\item $\varepsilon_i(t)$: multivariate random effect, $\varepsilon(t) \sim \mathcal{N}(0,\Sigma)$.
\end{itemize}
    \item \textbf{Excess mortality}:%\\[-0.5cm]
$$E_{irs}(t) := y_{irs}(t) - \tilde{y}_{irs}(t).$$
where $\tilde{y}_{irs}(t)$ is the expected number of deaths (i.e., posterior predictive distribution after removing $\varepsilon_i(t)$).
\end{itemize}
}


\headerbox{Temporal trends and cross-cause correlations}{name=res1,column=0,span=3,below=model, above=bottom}{
\begin{itemize}[leftmargin=1em]
\compresslist
\item The model captures seasonal and long-term trends of cause-specific mortality in 2011-2019 (see Fig 1 for 80+).
\item Expected mortality peaks between January and February for every cause.
\item Shift in peak timing during COVID-19.
\item Strong pairwise correlations in cause-specific excess mortality.
\end{itemize}

\noindent
\begin{minipage}[t]{0.55\linewidth}
 \vspace{0pt}
\textbf{Cause-specific excess by phase:}\\[-0.6cm]
\begin{itemize}[leftmargin=1em]
\compresslist
\item CVD mortality excess concentrates during COVID-19 wave, followed by deficits (see Fig 3 for 80+).
\item Strong mortality decrease for mental/neurological disorders (-12\%), especially during the Alpha wave (-24\%) in 80+.
\item Cancer mortality deficit mainly during the Alpha wave in 80+.
\item Respiratory (non-COVID) mortality consistently below expected across all ages, with larger deficits at older ages.
\end{itemize}
\end{minipage}
\hspace{1.6em}
\begin{minipage}[t]{0.38\linewidth}
 \vspace{0pt}
\textbf{Cancer exploration}\\[-0.6cm]
\begin{itemize}[leftmargin=1em]
\compresslist
\item Deficit in cancer deaths investigated using multiple-cause-of-death data.
\item Total deaths with cancer did not decline in Jan–Feb 2021; slight increase in 65–79, larger in 80+.
\item Largest increase observed for high-survival cancers, particularly in the 80+ age group.
\end{itemize}

\headerbox{Correlation between excess mortalities}{name=res4,span=3,column=3,below=res3}{
\begin{itemize}[leftmargin=1em]
\compresslist
\item Strong positive partial correlations (up to ~0.5) between COVID-19 excess mortality and excess deaths from CVD and mental/neurological diseases.
\item Similar positive correlations with other respiratory diseases.
\item Correlations peak at lags of 1–3 weeks, indicating COVID-19 excess deaths occurred slightly after CVD and mental/neurological deaths.
\item Cancer shows weak or negative correlations with COVID-19 mortality.
\end{itemize}
